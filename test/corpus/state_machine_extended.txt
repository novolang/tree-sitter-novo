state_machine — the output channel, fields, rows with bodies and a timeout
================================================================================

state_machine Pairing(out: Outbox)
    fields
        var mitm: Bool
    states
        Idle
        WaitConfirm
        Failed
    events
        Request(auth: Int)
        Confirm(v: Int)
        Timeout
    transitions
        (Idle, Request(auth)) =>
            self.mitm = auth > 0
            out.sent = out.sent + 1
            WaitConfirm after 30 s Timeout
        (WaitConfirm, Confirm(v)) if v == 0 => Failed
        (WaitConfirm, Timeout) => Failed
        (_, Timeout) => _
    enter Failed
        out.sent = out.sent + 100
    exit WaitConfirm
        out.sent = out.sent + 1

        On
        Broken
    events
        Press
        Smash
    transitions
        (Off, Press) => On
        (Off | On, Smash) => Broken
        (Broken, Smash | Press) => Off
